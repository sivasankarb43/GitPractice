	public static void stringReverse(String name) {

		// method 1 using charAt

		String reversed1 = "";

		for (int i = name.length() - 1; i >= 0; i--) {

			reversed1 = reversed1 + name.charAt(i);
		}
		System.out.println("method 1: " + reversed1);

		// method 2 using toCharArray

		char[] charName = name.toCharArray();

		String reversed2 = "";

		for (int i = charName.length - 1; i >= 0; i--) {

			reversed2 = reversed2 + charName[i];

		}

		System.out.println("method 2: " + reversed2);

		// method 3

		StringBuilder sb = new StringBuilder(name);
		sb.reverse();
		System.out.println("method 3: " + sb);

		// method 4

		StringBuffer sf = new StringBuffer(name);
		sf.reverse();
		System.out.println("method 4: " + sf);

	}

	public static void reverseNum(int a) {

		int rev = 0;

		while (a != 0) {

			rev = rev * 10;
			rev = rev + a % 10;
			a = a / 10;

		}

		System.out.println("Reversed Number: " + rev);

	}

	public static void numberIsArmstrong(int num) {

		int count = 0;
		int temp = num;
		int armstrg = 0;
		while (temp > 0) {

			temp = temp / 10;
			count++;
		}

		temp = num;
		while (temp > 0) {

			int last = temp % 10;
			armstrg = (int) (armstrg + (Math.pow(last, count)));
			temp = temp / 10;

		}

		if (armstrg == num) {

			System.out.println("The given number is armstrong");

		} else {
			System.out.println("The given number is not armstrong");
		}

	}

	public static void findTheGivenNumberIsPrime(int num) {

		boolean isPrime = true;
		if (num == 0 || num == 1) {
			isPrime = false;
		} else {
			for (int i = 2; i <= num / 2; i++) {

				if (num % i == 0) {
					isPrime = false;
					System.out.println("The given number is not prime");
					break;
				}

			}
		}
		if (isPrime) {
			System.out.println("The given number prime");
		}

	}

	public static void findFirstnonrepCharinString(String word) {

		LinkedHashMap<Character, Integer> map = new LinkedHashMap<>();
		boolean present = false;

			map.put(c, map.getOrDefault(c, 0) + 1);

		}

		for (Map.Entry<Character, Integer> entry : map.entrySet()) {

			if (entry.getValue() == 1) {
				present = true;
				System.out.println("The first non-repetative character in the given word is: " + entry.getKey());
				break;
			}
		}
		if (!present) {
			System.out.println("There is no non-repetative character in the given word");
		}

		// method 2

		for (char c : word.toCharArray()) {

			if (word.indexOf(c) == word.lastIndexOf(c)) {
				System.out.println("The first non-repetative character in the given word is: " + c);
				break;
			}

		}
	}

	public static void findMissingNum(int[] arr) {

		int n = arr.length + 1;
		int expected = 0;
		int sum = 0;
		for (int i : arr) {
			sum += i;
		}

	}
	
	public static void secondLargest(int[] arr) {
		
		int first = Integer.MIN_VALUE;
		int second = Integer.MIN_VALUE;
		int third = Integer.MIN_VALUE;
		
		for(int i : arr) {
			
			if(i>first) {
				third=second;
				second=first;
				first=i;
			}else if(i>second) {
				third=second;
				second=i;
			}else if(i>third) {
				third=i;
			}
			
		}
		
		System.out.println("The second largest value is: "+second);
		System.out.println("The third largest value is: "+third);
		
	}
	
	public static void pyramid(int size) {
		
		for(int i=1; i<=size; i++) {
			
			for(int j =1; j<=size-i; j++) {
				System.out.print(" ");
			}
			for(int j =1; j<=i; j++) {
				System.out.print(" *");
			}
			System.out.println();
			
		}
